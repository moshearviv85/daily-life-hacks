# Note on precedence: a crawler obeys ONLY the most specific group that names
# it — it does not merge the `*` group in. So every named group below repeats
# the same Disallow lines. Omitting them is how a named bot accidentally gets
# access the wildcard group denies.

User-agent: *
Allow: /
Disallow: /admin/
Disallow: /.git/
Disallow: /node_modules/

Disallow: /admin/
